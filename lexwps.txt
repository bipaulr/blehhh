PSEUDOCODE: LEXICAL ANALYZER FROM TERMINAL INPUT

BEGIN
    SET keywords = {int, float, char, if, else, while, for, return}

    WHILE READ_CHARACTER(terminal) gives ch other than EOF
        IF ch is whitespace
            CONTINUE
        END IF

        IF ch is a letter OR underscore
            SET word = empty string
            WHILE ch is a letter, digit or underscore AND length(word) < 63
                APPEND ch TO word
                SET ch = READ_CHARACTER(terminal)
            END WHILE
            PUT_BACK ch INTO terminal input if it is not EOF

            IF word is in keywords
                PRINT word, "Keyword"
            ELSE
                PRINT word, "Identifier"
            END IF
        ELSE IF ch is a digit
            SET word = empty string, has_dot = FALSE
            WHILE (ch is a digit OR (ch is '.' AND NOT has_dot)) AND length(word) < 63
                IF ch is '.'
                    SET has_dot = TRUE
                END IF
                APPEND ch TO word
                SET ch = READ_CHARACTER(terminal)
            END WHILE
            PUT_BACK ch INTO terminal input if it is not EOF
            PRINT word, "Number"
        ELSE IF ch is a double quote
            SET word = ch
            WHILE READ_CHARACTER(terminal) gives ch other than a double quote or EOF
                APPEND ch TO word
            END WHILE
            IF the closing double quote was not found
                PRINT "Unterminated or too long string literal"
                STOP with failure
            END IF
            APPEND the closing double quote TO word
            PRINT word, "String"
        ELSE IF ch is one of + - * / % = < > !
            SET next = READ_CHARACTER(terminal)
            IF ch is one of < > = ! AND next is '='
                PRINT ch followed by '=', "Operator"
            ELSE
                PUT_BACK next INTO terminal input if it is not EOF
                PRINT ch, "Operator"
            END IF
        ELSE IF ch is one of ; , ( ) { }
            PRINT ch, "Delimiter"
        ELSE
            PRINT ch, "Unknown"
        END IF
    END WHILE
END

NOTE: On Ubuntu, Ctrl+D on an empty line signals EOF.
This version stores at most 63 characters per token. Longer runs are split
into separate tokens, matching lexw.c; it has no long-token error check.
