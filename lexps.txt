PSEUDOCODE: LEXICAL ANALYZER FROM A FILE

BEGIN
    SET keywords = {int, float, char, if, else, while, for, return}

    IF more than one filename argument is supplied
        PRINT usage message
        STOP with failure
    END IF

    IF a filename argument is supplied
        SET filename = argument
    ELSE
        PRINT "Enter source filename: "
        READ filename from terminal
        REMOVE trailing newline characters
    END IF

    OPEN filename FOR reading AS source
    IF opening fails
        PRINT file-opening error
        STOP with failure
    END IF

    WHILE READ_CHARACTER(source) gives ch other than EOF
        IF ch is whitespace
            CONTINUE
        ELSE IF ch is a letter OR underscore
            SET word = empty string
            WHILE ch is a letter, digit or underscore AND length(word) < 63
                APPEND ch TO word
                SET ch = READ_CHARACTER(source)
            END WHILE
            IF ch is still a letter, digit or underscore
                PRINT "Identifier exceeds 63 characters"
                CLOSE source
                STOP with failure
            END IF
            IF ch is not EOF
                PUT_BACK ch INTO source
            END IF
            IF word is in keywords
                PRINT word, "Keyword"
            ELSE
                PRINT word, "Identifier"
            END IF
        ELSE IF ch is a digit
            SET word = empty string, has_dot = FALSE
            WHILE (ch is a digit OR (ch is '.' AND NOT has_dot)) AND length(word) < 63
                IF ch is '.'
                    SET has_dot = TRUE
                END IF
                APPEND ch TO word
                SET ch = READ_CHARACTER(source)
            END WHILE
            IF ch is still a digit
                PRINT "Number exceeds 63 characters"
                CLOSE source
                STOP with failure
            END IF
            IF ch is not EOF
                PUT_BACK ch INTO source
            END IF
            PRINT word, "Number"
        ELSE IF ch is a double quote
            SET word = ch
            WHILE READ_CHARACTER(source) gives ch other than a double quote or EOF
                APPEND ch TO word
            END WHILE
            IF the closing double quote was not found
                PRINT "Unterminated or too long string literal"
                CLOSE source
                STOP with failure
            END IF
            APPEND the closing double quote TO word
            PRINT word, "String"
        ELSE IF ch is one of + - * / % = < > !
            SET next = READ_CHARACTER(source)
            IF ch is one of < > = ! AND next is '='
                PRINT ch followed by '=', "Operator"
            ELSE
                IF next is not EOF
                    PUT_BACK next INTO source
                END IF
                PRINT ch, "Operator"
            END IF
        ELSE IF ch is one of ; , ( ) { }
            PRINT ch, "Delimiter"
        ELSE
            PRINT ch, "Unknown"
        END IF
    END WHILE

    IF a file-reading error occurred
        PRINT reading error
        CLOSE source
        STOP with failure
    END IF
    CLOSE source
END
