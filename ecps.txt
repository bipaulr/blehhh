PSEUDOCODE: EPSILON CLOSURE

BEGIN
    READ number_of_states, number_of_transitions
    VALIDATE 1 <= number_of_states <= 20 AND number_of_transitions >= 0
    IF reading or validation fails
        PRINT error AND STOP with failure
    END IF

    INITIALIZE reach[][] to FALSE

    FOR each transition
        READ source, symbol, destination
        VALIDATE source and destination are in 0 TO number_of_states - 1
        IF reading or validation fails
            PRINT error AND STOP with failure
        END IF
        IF symbol = 'e'
            SET reach[source][destination] = TRUE
        END IF
    END FOR

    FOR i = 0 TO number_of_states - 1
        SET reach[i][i] = TRUE
    END FOR

    FOR k = 0 TO number_of_states - 1
        FOR i = 0 TO number_of_states - 1
            FOR j = 0 TO number_of_states - 1
                IF reach[i][k] AND reach[k][j]
                    SET reach[i][j] = TRUE
                END IF
            END FOR
        END FOR
    END FOR

    FOR each state i
        PRINT "e-closure of state", i, "= {"
        FOR each state j
            IF reach[i][j]
                PRINT j
            END IF
        END FOR
        PRINT "}"
    END FOR
END
